	import cv2

cap = cv2.VideoCapture('video.mp4')
frames = []

while True:
    ret, frame = cap.read()
    if not ret:
        break
    frames.append(frame)
cap.release()

for frame in reversed(frames):
    cv2.imshow('Reverse Playback', frame)
    if cv2.waitKey(25) & 0xFF == ord('q'):
        break

cv2.destroyAllWindows()
