import cv2
import numpy as np

img = cv2.imread('image.jpg')
(h, w) = img.shape[:2]

M = np.float32([[1, 0, 100], [0, 1, 50]])  # shift x=100, y=50
shifted = cv2.warpAffine(img, M, (w, h))

cv2.imshow('Original', img)
cv2.imshow('Translated', shifted)
cv2.waitKey(0)
cv2.destroyAllWindows()
